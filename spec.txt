1. Общая идея
Ядро (TrendPulse-Core) должно не знать, какая именно модель/провайдер используется.
Вся работа с LLM идёт через:

единый LLM Gateway (или адаптер),
а «модули» – это плагины к этому Gateway:
– OpenRouter,
– Google Gemini,
– GLM / Qwen / DeepSeek,
– OpenAI / Anthropic,
– Together / Fireworks / Groq и т.д.

5. Как это выглядит в коде (пример интерфейса)
Условный интерфейс:

python

# core/llm/backends.py
from abc import ABC, abstractmethod
from typing import AsyncIterator, List, Dict, Any

class LLMBackend(ABC):
    @abstractmethod
    async def chat(
        self,
        messages: List[Dict[str, Any]],
        *,
        model: str | None = None,
        temperature: float = 0.7,
        max_tokens: int = 2048,
    ) -> str: ...

    @abstractmethod
    async def stream_chat(
        self,
        messages: List[Dict[str, Any]],
        *,
        model: str | None = None,
        temperature: float = 0.7,
        max_tokens: int = 2048,
    ) -> AsyncIterator[str]: ...

    @abstractmethod
    async def embed(
        self,
        texts: List[str],
        *,
        model: str | None = None,
    ) -> List[List[float]]: ...
Модули:

python

# modules/openrouter/backend.py
from openrouter import OpenRouter  # условно
from core.llm.backends import LLMBackend

class OpenRouterBackend(LLMBackend):
    def __init__(self, api_key: str, base_url: str | None = None):
        self.client = OpenRouter(api_key=api_key, base_url=base_url)

    async def chat(self, messages, *, model=None, temperature=0.7, max_tokens=2048) -> str:
        resp = await self.client.chat.send(
            model=model or "qwen/qwen-2.5-72b-instruct",
            messages=messages,
            temperature=temperature,
            max_tokens=max_tokens,
        )
        return resp.text

    # ... stream_chat, embed
